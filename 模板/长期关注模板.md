<%*
const rawTitle = await tp.system.prompt("长期关注事项标题");
if (!rawTitle || !rawTitle.trim()) {
  new Notice("已取消：标题不能为空");
  return;
}
const title = rawTitle.trim();
const cleanSegment = (value) => value.replace(/[\\:*?"<>|]/g, "-").trim();
const cleanedTitle = cleanSegment(title.replace(/\//g, "-"));
const fileTitle = cleanedTitle || ("未命名-" + tp.date.now("YYYYMMDD-HHmmss"));
const rawCategory = await tp.system.prompt("分类（例如：宠物、工作/雪球，可留空）");
const category = (rawCategory || "")
  .split(/[\\/]/)
  .map(cleanSegment)
  .filter(Boolean)
  .join("/");
const rawTags = await tp.system.prompt("标签（多个标签用逗号分隔，可留空）");
const tags = (rawTags || "")
  .split(/[,，]/)
  .map((item) => item.trim())
  .filter(Boolean);
const source = (await tp.system.prompt("来源（AI、网址、书籍或个人记录，可留空）") || "").trim();
const date = tp.date.now("YYYY-MM-DD");
const yaml = (value) => JSON.stringify(value);
const tagsYaml = tags.length
  ? "tags:\n" + tags.map((item) => "  - " + yaml(item)).join("\n")
  : "tags: []";
const targetFolder = "长期关注" + (category ? "/" + category : "");
const ensureFolder = async (folderPath) => {
  const parts = folderPath.split("/").filter(Boolean);
  let current = "";
  for (const part of parts) {
    current = current ? current + "/" + part : part;
    if (!app.vault.getAbstractFileByPath(current)) {
      await app.vault.createFolder(current);
    }
  }
};
await ensureFolder(targetFolder);
await tp.file.move(targetFolder + "/" + fileTitle);
tR += [
  "---",
  "类型: 长期关注",
  "状态: 有效",
  tagsYaml,
  "来源: " + (source ? yaml(source) : ""),
  "创建日期: " + date,
  "更新日期: " + date,
  "---",
  "",
  "# " + title,
  "",
  "## 当前情况",
  "",
  "## 持续目标",
  "",
  "## 下一步",
  "",
  "- [ ] ",
  "",
  "## 记录",
  "",
  "### " + date,
  ""
].join("\n");
%>
