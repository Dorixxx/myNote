<%*
const rawTitle = await tp.system.prompt("操作方法标题");
if (!rawTitle || !rawTitle.trim()) {
  new Notice("已取消：标题不能为空");
  return;
}
const title = rawTitle.trim();
const cleanedTitle = title.replace(/[\\/:*?"<>|]/g, "-").trim();
const fileTitle = cleanedTitle || ("未命名-" + tp.date.now("YYYYMMDD-HHmmss"));
const rawTopics = await tp.system.prompt("主题（多个主题用逗号分隔，可留空）");
const enteredTopics = (rawTopics || "")
  .split(/[,，]/)
  .map((item) => item.trim())
  .filter(Boolean);
const topics = [...new Set([].concat(enteredTopics))];
const source = (await tp.system.prompt("来源（AI、网址、书籍或个人记录，可留空）") || "").trim();
const date = tp.date.now("YYYY-MM-DD");
const yaml = (value) => JSON.stringify(value);
const topicsYaml = topics.length
  ? "tags:\n" + topics.map((item) => "  - " + yaml(item)).join("\n")
  : "tags: []";
await tp.file.move("知识库/操作指南/" + fileTitle);
tR += [
  "---",
  "类型: 操作方法",
  "状态: 待验证",
  topicsYaml,
  "来源: " + (source ? yaml(source) : ""),
  "创建日期: " + date,
  "更新日期: " + date,
  "---",
  "",
  "# " + title,
  "",
  "## 适用场景",
  "",
  "## 准备条件",
  "",
  "## 操作步骤",
  "",
  "1. ",
  "",
  "2. ",
  "",
  "## 验证方法",
  "",
  "## 常见问题",
  "",
  "## 来源",
  ""
].join("\n");
%>
